\begin{itemize}
  \item[(a)] We have
    \begin{equation*}
      \begin{split}
        \hat{p}(X) &= \frac{e^{\hat{\beta}_0 + \hat{\beta}_1 X_1 + \hat{\beta}_2 X_2}}
        {1 + e^{\hat{\beta}_0 + \hat{\beta}_1 X_1 + \hat{\beta}_2 X_2}} \\
        &= \frac{e^{-6 + 0.05 X_1 + X_2}}{1 + e^{-6 + 0.05 X_1 + X_2}} \\
        &= \frac{e^{-6 + 0.05 (40) + 3.5}}{1 + e^{-6 + 0.05 (40) + 3.5}} \\
        &\approx 0.377540668798145,
      \end{split}
    \end{equation*}
    where we make use of the following calculation:
  \scriptsize\begin{verbatim}
> p_hat <- function(x_1, x_2) {
+   exp(-6 + 0.05 * x_1 + x_2) / (1 + exp(-6 + 0.05 * x_1 + x_2))
+ }
> p_hat(40, 3.5)
[1] 0.3775407
  \end{verbatim}\normalsize
  \item[(b)] Suppose we have $\hat{p}(X) = 0.5$, then we must have
    \begin{equation*}
      \begin{split}
        \hat{p}(X) = \frac{e^{-6 + 0.05 X_1 + 3.5}}{1 + e^{-6 + 0.05 X_1 + 3.5}}
        &= 0.5 \\
        \Leftrightarrow \ln{\biggl(\frac{e^{-6 + 0.05 X_1 + 3.5}}
        {1 + e^{-6 + 0.05 X_1 + 3.5}}\biggr)} &= \ln{0.5} \\
        \Leftrightarrow \ln{(e^{-6 + 0.05 X_1 + 3.5})}
        - \ln{(1 + e^{-6 + 0.05 X_1 + 3.5})} &= \ln{0.5} \\
        \Leftrightarrow \ln{(e^{-6 + 0.05 X_1 + 3.5})}
        - \ln{1}\cdot\ln{e^{-6 + 0.05 X_1 + 3.5})} &= \ln{0.5} \\
        \Leftrightarrow \ln{(e^{-6 + 0.05 X_1 + 3.5})} &= \ln{0.5} \\
        \Leftrightarrow -6 + 0.05 X_1 + 3.5 &= \ln{0.5} \\
        \Leftrightarrow 0.05 X_1 &= \ln{0.5} + 6 - 3.5 \\
        \Leftrightarrow X_1 &= \frac{\ln{0.5} + 2.5}{0.05} \\
        &= 20\ln{0.5} + 50 \\
        &\approx 36.13706,
      \end{split}
    \end{equation*}
    where we make use of the following calculation:
  \scriptsize\begin{verbatim}
> x_1 <- 20 * log(0.5) + 50
> print(x_1)
[1] 36.13706
  \end{verbatim}\normalsize
    Thus we conclude that the student would need to study for approximately 37
    hours to have a 50\% change of getting an A in the class.
\end{itemize}
