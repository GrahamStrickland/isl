\begin{itemize}
  \item[(a)] If the odds of defaulting are given by 0.37, then we have
    $p(X) / [1 - p(X)] = 0.37$, so that we have the following:
    \begin{equation*}
      \begin{split}
        \frac{p(X)}{1 - p(X)} &= 0.37 \\
        \Leftrightarrow p(X) &= 0.37(1 - p(X)) \\
        \Leftrightarrow 1.37p(X) &= 0.37 \\
        \Leftrightarrow p(X) &\approx 0.270073,
      \end{split}
    \end{equation*}
    so that approximately 27\% of the people will default.
  \item[(b)] Using the same logic as in (a), we have
    \begin{equation*}
      \begin{split}
        \frac{p(X)}{1 - p(X)} &= \frac{0.16}{1 - 0.16} \\
        &= \frac{0.16}{0.84} \\
        &\approx 0.1904762,
      \end{split}
    \end{equation*}
    so that the odds that she will default are approximately 0.19.
\end{itemize}
